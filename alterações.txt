SELECT
	e.nome_fantasia,
	e.data_inicio_atividade,
	c.descricao,
	e.logradouro,
	e.bairro,
	m.descricao,
	e.ddd_1,
	e.telefone_1,
	e.correio_eletronico
FROM estabelecimento e
INNER JOIN cnaes c ON e.cnae_fiscal_principal = c.codigo
INNER JOIN municipios m ON e.municipio = m.codigo
WHERE
	e.municipio = '9571'
AND
	e.situacao_cadastral = '02'
AND
	e.data_inicio_atividade >= '20260901'
AND
	e.nome_fantasia IS NOT NULL
ORDER BY
	e.data_inicio_atividade DESC
--LIMIT 10
;
